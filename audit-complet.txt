  async function runEvaluationOnExtraction(extractedVehicle: VehicleExtraction) {
    const evalRes = await fetch('/api/evaluate', {

    const rawEvalText = await evalRes.text();
    const evalData = rawEvalText ? JSON.parse(rawEvalText) : {};
    if (!evalRes.ok) {
      throw new Error(
        evalData.error || `Erreur /api/evaluate (HTTP ${evalRes.status})`
      );

function parseReadingDateToTimestamp(rawDate: string | null): number | null {
  if (!rawDate) return null;
  const trimmed = rawDate.trim();

  // YYYY-MM-DD
  if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) {
    const ts = Date.parse(trimmed);
    return Number.isNaN(ts) ? null : ts;
  }

  // YYYY-MM
  if (/^\d{4}-\d{2}$/.test(trimmed)) {
    const ts = Date.parse(`${trimmed}-01`);
    return Number.isNaN(ts) ? null : ts;
  }

  // DD/MM/YYYY
  const dmy = trimmed.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
  if (dmy) {
    const ts = Date.parse(`${dmy[3]}-${dmy[2]}-${dmy[1]}`);
    return Number.isNaN(ts) ? null : ts;
  }

  // MM/YYYY
  const my = trimmed.match(/^(\d{2})\/(\d{4})$/);
  if (my) {
    const ts = Date.parse(`${my[2]}-${my[1]}-01`);
    return Number.isNaN(ts) ? null : ts;
  }

  // YYYY
  if (/^\d{4}$/.test(trimmed)) {
    const ts = Date.parse(`${trimmed}-06-01`);
    return Number.isNaN(ts) ? null : ts;
  }

  return null;
}

export function evaluateMileageConsistency(

  const km = extraction.kilometrage_km;

    return {
      status: 'unknown',
      expectedAnnualKm: null,
      observedAnnualKm: null,
      note: null,
    };
  }

  const now = new Date();
  const currentYear = now.getFullYear();
  const currentMonth = now.getMonth() + 1;
  const monthMec = extraction.mois_mec ?? 6;

  const ageMonths = Math.max(1, (currentYear - year) * 12 + (currentMonth - monthMec));
  const ageYears = ageMonths / 12;
  const observedAnnualKm = Math.round(km / ageYears);
  const expectedAnnualKm = 13500;

  if (ageYears >= 4 && observedAnnualKm < 4000) {

      status: 'suspicious_low',

      observedAnnualKm,

        'fr-FR'

        'fr-FR'
      )} km/an) : vÃ©rifier la cohÃ©rence avec les rapports Histovec et CT.`,
    };

  if (km >= 160000 || observedAnnualKm >= 28000) {

export async function normalizeMobileFile(file: File): Promise<File> {
  const isHeic =
    file.type === 'image/heic' ||
    file.type === 'image/heif' ||
    /\.(heic|heif)$/i.test(file.name);

  if (!isHeic) return file;

  // Garde-fou OOM mobile : rejeter les fichiers HEIC/RAW massifs (> 25 Mo)
  if (file.size > 25 * 1024 * 1024) {
    throw new Error(
      'Fichier HEIC trop volumineux (> 25 Mo). Compresse-le ou capture une photo plus lÃ©gÃ¨re.'
    );
  }

  const heic2any = (await import('heic2any')).default;
  const convertedBlob = await heic2any({
    blob: file,
    toType: 'image/jpeg',
    quality: 0.75,
  });

  const singleBlob = Array.isArray(convertedBlob) ? convertedBlob[0] : convertedBlob;

  });

function blobToDataUrl(blob: Blob): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => {
      if (typeof reader.result === 'string') {
        resolve(reader.result);
      } else {
        reject(new Error('Impossible de lire le fichier image.'));
      }
    };
    reader.onerror = () => reject(new Error('Erreur de lecture du fichier.'));
    reader.readAsDataURL(blob);
  });
}

async function extractTextFromPdf(file: File): Promise<{
  text: string;
  pageCount: number;
}> {
  const pdfjsLib = await import('pdfjs-dist');

  if (typeof window !== 'undefined' && pdfjsLib.GlobalWorkerOptions) {
    pdfjsLib.GlobalWorkerOptions.workerSrc = `https://unpkg.com/pdfjs-dist@${pdfjsLib.version}/build/pdf.worker.min.mjs`;
  }

  const arrayBuffer = await file.arrayBuffer();
  const loadingTask = pdfjsLib.getDocument({ data: arrayBuffer });
  const pdf = await loadingTask.promise;
  const pageCount = pdf.numPages;

  const pagesText: string[] = [];
  const maxPages = Math.min(pageCount, 20);

  for (let i = 1; i <= maxPages; i++) {
    const page = await pdf.getPage(i);
    const content = await page.getTextContent();
    const strings = content.items
      .map((item) => ('str' in item ? item.str : ''))
      .filter(Boolean);
    pagesText.push(`--- Page ${i} ---\n${strings.join(' ')}`);
  }

  return {
    text: pagesText.join('\n\n'),
    pageCount,
  };
}

export async function processUploadedFile(file: File): Promise<ProcessedUpload> {
  const isPdf =
    file.type === 'application/pdf' || /\.pdf$/i.test(file.name);

  if (isPdf) {
    const { text, pageCount } = await extractTextFromPdf(file);
    const sizeKb = Math.max(1, Math.round(file.size / 1024));
    return {
      kind: 'pdf',
      fileName: file.name,
      sizeKb,
      pdfText: text,
      pdfPageCount: pageCount,
    };
  }

  const normalizedFile = await normalizeMobileFile(file);

  const compressedFile = await imageCompression(normalizedFile, {
    maxSizeMB: 1.2,
    maxWidthOrHeight: 1600,
    useWebWorker: true,
    fileType: 'image/jpeg',
    initialQuality: 0.82,
  });

  const imageDataUrl = await blobToDataUrl(compressedFile);
  const sizeKb = Math.max(1, Math.round(compressedFile.size / 1024));

  return {
    kind: 'image',
    fileName: normalizedFile.name,
    sizeKb,
    imageDataUrl,
  };
}

function getSecret(): string {

}

export function createMagicToken(payload: MagicLeadPayload): string {
  const json = JSON.stringify(payload);
  const dataB64 = Buffer.from(json, 'utf8').toString('base64url');
  const sig = crypto
    .createHmac('sha256', getSecret())
    .update(dataB64)
    .digest('base64url');

  return `${dataB64}.${sig}`;
}

export function verifyMagicToken(token: string): MagicLeadPayload | null {
  if (!token || !token.includes('.')) return null;

  const [dataB64, sig] = token.split('.');
  if (!dataB64 || !sig) return null;

  const expectedSig = crypto
    .createHmac('sha256', getSecret())
    .update(dataB64)
    .digest('base64url');

  const sigBuf = Buffer.from(sig);
  const expectedBuf = Buffer.from(expectedSig);

  if (sigBuf.length !== expectedBuf.length) return null;
  if (!crypto.timingSafeEqual(sigBuf, expectedBuf)) return null;

  try {
    const raw = Buffer.from(dataB64, 'base64url').toString('utf8');
    const parsed = JSON.parse(raw) as MagicLeadPayload;
    if (!parsed.email || !parsed.vehicleHint) return null;
    return parsed;
  } catch {
    return null;
  }
}
